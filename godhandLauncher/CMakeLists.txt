cmake_minimum_required(VERSION 3.21)
project(GodHandLauncher LANGUAGES CXX)

find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBARCHIVE REQUIRED IMPORTED_TARGET libarchive)

add_library(godhand_installer STATIC src/installer.cpp)
target_compile_features(godhand_installer PUBLIC cxx_std_17)
target_include_directories(godhand_installer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(godhand_installer PUBLIC PkgConfig::LIBARCHIVE)

add_executable(godhand_iso_inspect src/inspect_main.cpp)
target_compile_features(godhand_iso_inspect PRIVATE cxx_std_17)
target_link_libraries(godhand_iso_inspect PRIVATE godhand_installer)

add_library(godhand_settings STATIC src/settings.cpp)
target_compile_features(godhand_settings PUBLIC cxx_std_17)
target_include_directories(godhand_settings PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")

option(GODHAND_BUILD_GTK_LAUNCHER "Build GTK3 desktop launcher" ON)
if(GODHAND_BUILD_GTK_LAUNCHER)
    pkg_check_modules(GTK3 QUIET IMPORTED_TARGET gtk+-3.0)
    pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
    if(GTK3_FOUND AND SDL2_FOUND)
        add_library(godhand_input STATIC src/input.cpp)
        target_link_libraries(godhand_input PUBLIC godhand_settings PkgConfig::SDL2)
        target_compile_features(godhand_input PUBLIC cxx_std_17)
        add_executable(GodHand_Recomp_Launcher src/main.cpp)
        target_compile_features(GodHand_Recomp_Launcher PRIVATE cxx_std_17)
        target_link_libraries(GodHand_Recomp_Launcher PRIVATE godhand_installer godhand_input godhand_settings PkgConfig::GTK3)
    else()
        message(WARNING "GTK3/SDL2 development files missing: launcher UI skipped. Install libgtk-3-dev libsdl2-dev.")
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(godhand_settings_tests tests/settings_tests.cpp)
    target_compile_features(godhand_settings_tests PRIVATE cxx_std_17)
    target_link_libraries(godhand_settings_tests PRIVATE godhand_settings)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(godhand_settings_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME godhand_settings_tests COMMAND godhand_settings_tests)
    add_executable(godhand_installer_tests tests/installer_tests.cpp)
    target_compile_features(godhand_installer_tests PRIVATE cxx_std_17)
    target_link_libraries(godhand_installer_tests PRIVATE godhand_installer)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(godhand_installer_tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME godhand_installer_tests COMMAND godhand_installer_tests)
endif()
